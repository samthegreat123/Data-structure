//2. Write a program to find out the factorial of a number using recursion  
(stack).  
#include<stdio.h>

long int factorial(int n)
{
    if(n == 0 || n == 1)
        return 1;
    else
        return n * factorial(n-1);
}

int main()
{
    int num;
    
    printf("Enter a number: ");
    scanf("%d",&num);

    printf("Factorial of %d is %ld", num, factorial(num));

    return 0;
}
