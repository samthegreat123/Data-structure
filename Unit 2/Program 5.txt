//5. Write a program to find the power of a given number using stack. 


#include <stdio.h>

#define MAX 100

int stack[MAX];
int top = -1;

void push(int value)
{
    stack[++top] = value;
}

int pop()
{
    return stack[top--];
}

int main()
{
    int base, power;
    int result = 1;

    printf("Enter base: ");
    scanf("%d", &base);

    printf("Enter power: ");
    scanf("%d", &power);

    for(int i = 0; i < power; i++)
    {
        push(base);
    }

    while(top != -1)
    {
        result = result * pop();
    }

    printf("Power = %d\n", result);

    return 0;
}
